src.data_processing

Streaming ingestion of the Monash environmental-sensor CSV (inside a ZIP).

The raw file has one CSV row per ingestion record. The measurements live in the jsondata column, which is a JSON array encoded as a string inside the CSV, e.g.::

[{"values": [{"time": 1702273965, "value": 23.8}],
  "variable": {"id": 68, "name": "Temperature", "unit": "°C"}}]

Design decisions (each is deliberate and tested in tests/test_parser.py):

  • Streaming. The ZIP member (~353 MB uncompressed) is read row by row with zipfile + csv.DictReader; it is never extracted to disk or loaded whole into memory.
  • Select variables by name, never by the nested id or array position.
  • Event time, not ingestion time. values[].time (Unix epoch seconds, a UTC instant) is the measurement time. createdate/processdate are pipeline metadata and are kept only for auditing.
  • No silent dropping. Every record or reading that cannot be used is counted in ParseStats under a named reason.
  • Deterministic duplicate rule. See deduplicate().

100def _to_float(x) -> float | None:
101    """Convert ``x`` to a finite float, or return ``None`` if that is impossible.
102
103    Booleans are rejected explicitly because ``float(True) == 1.0`` would
104    otherwise turn a JSON ``true`` into a plausible-looking reading.
105    """
106    if x is None or isinstance(x, bool):
107        return None
108    try:
109        f = float(x)
110    except (TypeError, ValueError):
111        return None
112    return f if math.isfinite(f) else None

115def parse_jsondata(
116    text: str | None,
117    variables: Iterable[str] | None,
118    stats: ParseStats | None = None,
119) -> list[tuple[str, str, int, float]]:
120    """Decode one ``jsondata`` string into readings for the requested variables.
121
122    Args:
123        text: The raw ``jsondata`` cell.
124        variables: Variable names to keep (exact match on ``variable.name``,
125            e.g. ``"Temperature"``). ``None`` keeps every variable.
126        stats: Optional counters, updated in place.
127
128    Returns:
129        A list of ``(name, unit, epoch_s, value)`` tuples. ``epoch_s`` is an
130        integer Unix time in seconds (UTC). Malformed input yields an empty
131        list and increments the relevant counter; it never raises.
132    """
133    stats = stats if stats is not None else ParseStats()
134    wanted = None if variables is None else set(variables)
135
136    if not text or not text.strip():
137        stats.bad_json += 1
138        return []
139    try:
140        payload = json.loads(text)
141    except json.JSONDecodeError:
142        stats.bad_json += 1
143        return []
144    if not isinstance(payload, list):
145        stats.not_a_list += 1
146        return []
147
148    out: list[tuple[str, str, int, float]] = []
149    for entry in payload:
150        var = entry.get("variable") if isinstance(entry, dict) else None
151        values = entry.get("values") if isinstance(entry, dict) else None
152        name = var.get("name") if isinstance(var, dict) else None
153        if not isinstance(name, str) or not isinstance(values, list):
154            stats.bad_entry += 1
155            continue
156        unit = var.get("unit") if isinstance(var.get("unit"), str) else ""
157        stats.variables_seen[(name, unit)] += len(values)
158        if wanted is not None and name not in wanted:
159            continue
160
161        for v in values:
162            if not isinstance(v, dict):
163                stats.bad_value += 1
164                continue
165            t = _to_float(v.get("time"))
166            if t is None or not (MIN_PLAUSIBLE_EPOCH <= t <= MAX_PLAUSIBLE_EPOCH):
167                stats.bad_time += 1
168                continue
169            val = _to_float(v.get("value"))
170            if val is None:
171                stats.bad_value += 1
172                continue
173            out.append((name, unit, int(t), val))
174            stats.readings_kept += 1
175    return out

178def find_csv_member(zf: zipfile.ZipFile) -> str:
179    """Return the name of the single CSV inside the archive.
180
181    Raises:
182        ValueError: if the archive holds zero or several CSV files, so the
183            caller must choose explicitly rather than us guessing.
184    """
185    csvs = [
186        n for n in zf.namelist()
187        if n.lower().endswith(".csv") and not n.startswith("__MACOSX/")
188    ]
189    if len(csvs) != 1:
190        raise ValueError(f"Expected exactly one CSV in archive, found: {csvs}")
191    return csvs[0]

194def iter_raw_rows(zip_path: str | Path, member: str | None = None) -> Iterator[dict]:
195    """Yield raw CSV rows (as dicts) from a ZIP without extracting it.
196
197    Args:
198        zip_path: Path to the ``.zip`` file.
199        member: CSV name inside the ZIP; detected automatically if omitted.
200    """
201    # jsondata cells can be long; lift the default 128 kB csv field limit.
202    csv.field_size_limit(min(sys.maxsize, 2**31 - 1))
203    with zipfile.ZipFile(zip_path) as zf:
204        member = member or find_csv_member(zf)
205        with zf.open(member) as raw:
206            # utf-8-sig strips a byte-order mark if present; newline="" is
207            # required by the csv module for quoted fields containing newlines.
208            text = io.TextIOWrapper(raw, encoding="utf-8-sig", newline="")
209            yield from csv.DictReader(text)

212def stream_readings(
213    zip_path: str | Path,
214    variables: Iterable[str] | None = ("Temperature",),
215    sensors: Iterable[str] | None = None,
216    stats: ParseStats | None = None,
217    limit_rows: int | None = None,
218) -> Iterator[tuple]:
219    """Stream tidy readings from the raw ZIP.
220
221    Args:
222        zip_path: Path to ``5EnvSensor_MayToDec2024_180kRows.zip``.
223        variables: Variable names to keep; ``None`` keeps all.
224        sensors: Optional sensor IDs (as strings) to keep.
225        stats: Optional :class:`ParseStats`, updated in place.
226        limit_rows: Stop after this many CSV rows (for quick inspection).
227
228    Yields:
229        Tuples in :data:`READING_COLUMNS` order.
230    """
231    stats = stats if stats is not None else ParseStats()
232    sensor_set = None if sensors is None else {str(s) for s in sensors}
233    for row in iter_raw_rows(zip_path):
234        if limit_rows is not None and stats.rows >= limit_rows:
235            break
236        stats.rows += 1
237        sid = (row.get("sensorid") or "").strip()
238        if sensor_set is not None and sid not in sensor_set:
239            continue
240        for name, unit, t, val in parse_jsondata(row.get("jsondata"), variables, stats):
241            yield (row.get("id"), sid, name, unit, t, val, row.get("createdate"))

244def readings_to_frame(readings: Iterable[tuple]) -> pd.DataFrame:
245    """Build a DataFrame from reading tuples and add timezone-aware timestamps.
246
247    Adds ``time_utc`` (tz-aware UTC) and ``time_local`` (tz-aware
248    Australia/Melbourne, so AEST/AEDT is handled by the tz database rather
249    than a fixed +10/+11 offset).
250    """
251    df = pd.DataFrame(list(readings), columns=READING_COLUMNS)
252    df["sensorid"] = df["sensorid"].astype(str)
253    df["epoch_s"] = df["epoch_s"].astype("int64")
254    df["value"] = df["value"].astype("float64")
255    df["time_utc"] = pd.to_datetime(df["epoch_s"], unit="s", utc=True)
256    df["time_local"] = df["time_utc"].dt.tz_convert(LOCAL_TZ)
257    return df

261class DedupReport:
262    """Outcome of :func:`deduplicate`.
263
264    Attributes:
265        rows_in: readings before deduplication.
266        rows_out: unique ``(sensorid, variable, epoch_s)`` keys after.

269        conflicting_keys: keys that had two or more *different* values.
270        max_conflict_spread: largest ``max - min`` among conflicting keys,
271            in the variable's own unit (0 if none), to judge whether the
272            conflicts matter.
273    """

275    rows_in: int
276    rows_out: int
277    exact_duplicates: int
278    conflicting_keys: int
279    max_conflict_spread: float

282def deduplicate(df: pd.DataFrame) -> tuple[pd.DataFrame, DedupReport]:
283    """Collapse readings to one value per ``(sensorid, variable, epoch_s)``.
284
285    Rule: identical repeats collapse to one; if a key has *different* values,
286    the **median** is used and the key is counted as a conflict. The median
287    is order-independent, so the result does not depend on CSV row order.
288    The conflict count and spread are reported so the rule can be revisited
289    if conflicts turn out to be common or large.
290
291    Returns:
292        ``(deduplicated_frame, DedupReport)``, sorted by sensor, variable and time.
293    """
294    key = ["sensorid", "variable", "epoch_s"]
295    exact = int(df.duplicated(subset=key + ["value"]).sum())
296    g = df.groupby(key, sort=True)
297    spread = g["value"].agg(lambda s: s.max() - s.min())
298    n_unique = g["value"].nunique()
299    conflicts = spread[n_unique > 1]
300
301    out = g.agg(
302        unit=("unit", "first"),
303        value=("value", "median"),
304        n_raw=("value", "size"),
305    ).reset_index()
306    out["time_utc"] = pd.to_datetime(out["epoch_s"], unit="s", utc=True)
307    out["time_local"] = out["time_utc"].dt.tz_convert(LOCAL_TZ)
308
309    report = DedupReport(
310        rows_in=len(df),
311        rows_out=len(out),
312        exact_duplicates=exact,
313        conflicting_keys=int(len(conflicts)),
314        max_conflict_spread=float(conflicts.max()) if len(conflicts) else 0.0,
315    )
316    return out, report

283def deduplicate(df: pd.DataFrame) -> tuple[pd.DataFrame, DedupReport]:
284    """Collapse readings to one value per ``(sensorid, variable, epoch_s)``.
285
286    Rule: identical repeats collapse to one; if a key has *different* values,
287    the **median** is used and the key is counted as a conflict. The median
288    is order-independent, so the result does not depend on CSV row order.
289    The conflict count and spread are reported so the rule can be revisited
290    if conflicts turn out to be common or large.
291
292    Returns:
293        ``(deduplicated_frame, DedupReport)``, sorted by sensor, variable and time.
294    """
295    key = ["sensorid", "variable", "epoch_s"]
296    exact = int(df.duplicated(subset=key + ["value"]).sum())
297    g = df.groupby(key, sort=True)
298    spread = g["value"].agg(lambda s: s.max() - s.min())
299    n_unique = g["value"].nunique()
300    conflicts = spread[n_unique > 1]
301
302    out = g.agg(
303        unit=("unit", "first"),
304        value=("value", "median"),
305        n_raw=("value", "size"),
306    ).reset_index()
307    out["time_utc"] = pd.to_datetime(out["epoch_s"], unit="s", utc=True)
308    out["time_local"] = out["time_utc"].dt.tz_convert(LOCAL_TZ)
309
310    report = DedupReport(
311        rows_in=len(df),
312        rows_out=len(out),
313        exact_duplicates=exact,
314        conflicting_keys=int(len(conflicts)),
315        max_conflict_spread=float(conflicts.max()) if len(conflicts) else 0.0,
316    )
317    return out, report
