src.features

Turn raw sensor readings into past-only forecasting examples.

The steps, in order, are:

  1. load_temperature() reads one sensor's deduplicated readings.
  2. to_grid() places them on a regular time grid. Empty slots stay NaN; nothing is interpolated, so no value is ever estimated from the future.
  3. make_supervised() builds lag columns and the future target.
  4. chronological_split() assigns examples to train, validation and test, removing any example whose time window would cross a partition boundary.

Every lag and target position is computed on the regular grid, so a row shift always equals a true time shift, even across gaps.

 35def load_temperature(
 36    csv_path: str | Path,
 37    sensorid: str,
 38    variable: str = "Temperature",
 39) -> pd.Series:
 40    """Load one sensor's readings from the tidy table written by the audit.
 41
 42    Args:
 43        csv_path: Path to ``data/processed/readings.csv``.
 44        sensorid: Sensor ID, e.g. ``"6012002000869"``.
 45        variable: Exact variable name to load.
 46
 47    Returns:
 48        Values in the variable's own unit, indexed by UTC instant and sorted by
 49        time, with any readings before :data:`ANALYSIS_START` removed.
 50
 51    Raises:
 52        ValueError: If the sensor and variable have no readings.
 53    """
 54    df = pd.read_csv(csv_path, usecols=["sensorid", "variable", "epoch_s", "value"],
 55                     dtype={"sensorid": str})
 56    df = df[(df["sensorid"] == str(sensorid)) & (df["variable"] == variable)]
 57    if df.empty:
 58        raise ValueError(f"No {variable!r} readings for sensor {sensorid} in {csv_path}")
 59    index = pd.to_datetime(df["epoch_s"], unit="s", utc=True)
 60    series = pd.Series(df["value"].to_numpy(), index=index, name=variable).sort_index()
 61    start = ANALYSIS_START.get(str(sensorid))
 62    return series if start is None else series[series.index >= pd.Timestamp(start)]

 65def to_grid(readings: pd.Series, step_min: int) -> pd.Series:
 66    """Resample readings onto a regular grid without using future information.
 67
 68    A slot labelled ``t`` holds the **latest reading in** ``(t - step, t]``. The
 69    slot is labelled at its end, so its value was already known at time ``t``.
 70    Labelling at the start of the slot would leak up to one step of the future
 71    into every feature.
 72
 73    Args:
 74        readings: Values indexed by a sorted, tz-aware UTC DatetimeIndex.
 75        step_min: Grid spacing in minutes. Must be a positive integer.
 76
 77    Returns:
 78        A regularly spaced series in which empty slots are NaN.
 79
 80    Raises:
 81        ValueError: If ``step_min`` is not positive or there are no readings.
 82    """
 83    if step_min <= 0:
 84        raise ValueError("step_min must be positive")
 85    if readings.empty:
 86        raise ValueError("Cannot build a grid from zero readings")
 87    return readings.resample(f"{step_min}min", closed="right", label="right",
 88                             origin="epoch").last()

 91def horizon_to_steps(horizon_min: int, step_min: int) -> int:
 92    """Convert a horizon in minutes to a whole number of grid steps.
 93
 94    Raises:
 95        ValueError: If the horizon is not a positive multiple of the step, since
 96            the target would otherwise not fall on the grid.
 97    """
 98    if horizon_min <= 0 or horizon_min % step_min:
 99        raise ValueError(f"Horizon {horizon_min} min must be a positive multiple "
100                         f"of the {step_min} min step")
101    return horizon_min // step_min

105    """Return feature names: ``lag_0`` is T(t), ``lag_1`` is T(t - step), and so on."""
106    return [f"lag_{i}" for i in range(n_lags)]

109def make_supervised(grid: pd.Series, n_lags: int, horizon_steps: int) -> pd.DataFrame:
110    """Build lagged features and the future target from a regular grid.
111
112    For issue time ``t`` the features are ``T(t), T(t - step), ...`` and the
113    target is ``T(t + horizon)``. Rows with any missing value are removed rather
114    than filled, so a gap can never be bridged.
115
116    Args:
117        grid: Output of :func:`to_grid`.
118        n_lags: Number of lag columns, at least 1.
119        horizon_steps: Steps between the issue time and the target, at least 1.
120
121    Returns:
122        A frame indexed by issue time with columns ``lag_0 ... lag_{n-1}`` and
123        ``target``.
124
125    Raises:
126        ValueError: If the arguments are invalid or the grid is irregular.
127    """
128    if n_lags < 1 or horizon_steps < 1:
129        raise ValueError("n_lags and horizon_steps must both be at least 1")
130    if len(grid) > 1 and grid.index.to_series().diff().dropna().nunique() != 1:
131        raise ValueError("Grid must be regularly spaced; use to_grid() first")
132    data = {name: grid.shift(i) for i, name in enumerate(lag_columns(n_lags))}
133    data["target"] = grid.shift(-horizon_steps)
134    return pd.DataFrame(data).dropna()

158def chronological_split(
159    frame: pd.DataFrame,
160    grid_index: pd.DatetimeIndex,
161    n_lags: int,
162    horizon_steps: int,
163) -> Split:
164    """Split examples by time, purging windows that cross a boundary.
165
166    The boundaries come from the grid's full time span, so they do not move when
167    the lag count or horizon changes. An example's window runs from its oldest
168    lag slot to its target slot, and the whole window must lie inside one
169    partition. That is the embargo: no value is shared between partitions.
170
171    Args:
172        frame: Output of :func:`make_supervised`.
173        grid_index: Index of the grid the frame was built from.
174        n_lags: Lag count used to build ``frame``.
175        horizon_steps: Horizon, in steps, used to build ``frame``.
176
177    Returns:
178        A :class:`Split` holding the masks and boundaries.
179    """
180    step = grid_index[1] - grid_index[0]
181    span = grid_index[-1] - grid_index[0]
182    b1 = grid_index[grid_index <= grid_index[0] + SPLIT_FRACTIONS[0] * span][-1]
183    b2 = grid_index[grid_index <= grid_index[0] + SPLIT_FRACTIONS[1] * span][-1]
184    first = frame.index - (n_lags - 1) * step
185    last = frame.index + horizon_steps * step
186    train = np.asarray(last <= b1)
187    val = np.asarray((first > b1) & (last <= b2))
188    test = np.asarray(first > b2)
189    purged = int(len(frame) - train.sum() - val.sum() - test.sum())
190    return Split(train, val, test, b1, b2, purged)
